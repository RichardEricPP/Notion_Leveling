snippet filter

	  filter(${2:condicion})
	$0

snippet select
	${1:base} |>
	  select(${2:variable_1}, ${3:variable_2}, ${4:variable_3})
	$0

snippet arrange
	${1:base} |>
	  arrange(desc(${2:variable}))
	$0

snippet summarise
	${1:base} |>
	  summarise(
	    n = n(),
	    promedio = mean(${2:variable}, na.rm = TRUE),
	    desviacion = sd(${2:variable}, na.rm = TRUE)
	  )
	$0

snippet group_by
	total <- nrow(${1:base})

	${1:base} |>
	  group_by(${2:grupo}) |>
	  summarise(
	    n = n(),
	    promedio = mean(${3:variable}, na.rm = TRUE),
	    desviacion = sd(${3:variable}, na.rm = TRUE)
	  ) |>
	  mutate(
	    porcentaje = n / total * 100
	  )
	$0

snippet mutate
	${1:base} |>
	  mutate(
	    ${2:nueva_variable} = ${3:formula}
	  )
	$0

snippet count
	${1:base} |>
	  count(${2:variable}) |>
	  mutate(
	    porcentaje = n / sum(n) * 100
	  )
	$0

snippet tabyl

	$0

# t.test: 2 grupos
snippet t_test
	t.test(
	  ${1:variable_numerica} ~ ${2:grupo},
	  data = ${3:base}
	)
	$0

# t.test: una media vs valor
snippet t_mu
	t.test(
	  ${1:base}[["${2:variable_numerica}"]],
	  mu = ${3:valor}
	)
	$0

# prop.test
snippet prop_test
	x <- sum(
	  ${1:base}[["${2:variable}"]] == "${3:valor}",
	  na.rm = TRUE
	)
	n <- nrow(${1:base})

	prop.test(
	  x = x,
	  n = n,
	  p = ${4:0.80}
	)
	$0

# cor.test — fórmula limpia, SIN $
snippet cor_test
	cor.test(
	  ~ ${1:variable_x} + ${2:variable_y},
	  data = ${3:base}
	)
	$0

# matriz de correlaciones + corrplot
snippet corrplot
	${1:matriz_cor} <- ${2:base} |>
	  select(
	    ${3:variable_1},
	    ${4:variable_2},
	    ${5:variable_3},
	    ${6:variable_4}
	  ) |>
	  cor(use = "complete.obs")

	corrplot(
	  ${1:matriz_cor},
	  method = "color",
	  addCoef.col = "black",
	  title = "${7:Matriz de correlaciones}"
	)
	$0

# lm
snippet lm
	${1:modelo} <- lm(
	  ${2:Y} ~ ${3:X},
	  data = ${4:base}
	)

	summary(${1:modelo})
	$0

snippet coord_polar
	${1:resumen} <- ${2:base} |>
	  count(${3:variable_cat}) |>
	  mutate(
	    porcentaje = n / sum(n) * 100
	  )

	ggplot(${1:resumen}, aes(x = "", y = porcentaje, fill = ${3:variable_cat})) +
	  geom_col(width = 1) +
	  coord_polar(theta = "y") +
	  geom_text(
	    aes(label = paste0(round(porcentaje, 1), "%")),
	    position = position_stack(vjust = 0.5)
	  ) +
	  labs(
	    title = "${4:Distribución porcentual}",
	    fill = "${5:Categoría}"
	  ) +
	  theme_void() +
	  theme(
	    plot.title = element_text(hjust = 0.5, face = "bold"),
	    legend.position = "bottom"
	  )
	$0
