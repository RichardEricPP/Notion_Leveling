# ============================================================
# 1. LIBRERÍAS
# ============================================================

snippet libra
	library(tidyverse)
	library(readxl)
	library(janitor)
	library(corrplot)
	$0


# ============================================================
# 2. IMPORTACIÓN
# ============================================================

snippet csv
	${1:datos} <- read_csv("${2:archivo.csv}")
	$0

snippet csv2
	${1:datos} <- read_csv2("${2:archivo.csv}")
	$0

snippet excel
	${1:datos} <- read_excel("${2:archivo.xlsx}")
	$0


# ============================================================
# 3. INSPECCIÓN DE LA BASE
# ============================================================

snippet mirar
	glimpse(${1:datos})
	summary(${1:datos})
	dim(${1:datos})
	names(${1:datos})
	$0

snippet nas
	colSums(is.na(${1:datos}))
	$0


# ============================================================
# 4. FILTRAR / SELECCIONAR / ORDENAR
# ============================================================

snippet filtra
	${1:datos} |>
	  filter(${2:condicion})
	$0

snippet selec
	${1:datos} |>
	  select(${2:variable_1}, ${3:variable_2}, ${4:variable_3})
	$0

snippet top
	${1:datos} |>
	  arrange(desc(${2:variable})) |>
	  head(${3:5})
	$0


# ============================================================
# 5. RESÚMENES / GROUP_BY / MUTATE / CATEGORÍAS
# ============================================================

snippet resu
	${1:datos} |>
	  summarise(
	    cantidad = n(),
	    promedio = mean(${2:variable_num}, na.rm = TRUE),
	    desviacion = sd(${2:variable_num}, na.rm = TRUE)
	  )
	$0

snippet resug
	${1:datos} |>
	  group_by(${2:grupo}) |>
	  summarise(
	    cantidad = n(),
	    promedio = mean(${3:variable_num}, na.rm = TRUE),
	    desviacion = sd(${3:variable_num}, na.rm = TRUE)
	  )
	$0

snippet filresu
	${1:datos} |>
	  filter(${2:condicion}) |>
	  summarise(
	    cantidad = n(),
	    promedio = mean(${3:variable_num}, na.rm = TRUE)
	  )
	$0

snippet muta
	${1:datos} |>
	  mutate(
	    ${2:nueva_variable} = ${3:formula}
	  )
	$0


snippet categorias
	${1:datos} |>
	  mutate(
	    ${2:nueva_categoria} = case_when(
	      ${3:condicion_1} ~ "${4:Categoria 1}",
	      ${5:condicion_2} ~ "${6:Categoria 2}",
	      ${7:condicion_3} ~ "${8:Categoria 3}",
	      TRUE ~ "${9:Otra}"
	    )
	  )
	$0


# ============================================================
# 6. PORCENTAJES / FRECUENCIAS
# ============================================================

snippet porcat
	${1:datos} |>
	  count(${2:variable_cat}) |>
	  mutate(
	    porcentaje = n / sum(n) * 100
	  )
	$0

snippet tabla
	tabyl(${1:datos}$${2:variable_cat})
	$0

snippet portotal
	total <- nrow(${1:datos})
	
	${1:datos} |>
	  filter(${2:condicion}) |>
	  summarise(
	    cantidad = n(),
	    porcentaje = cantidad / total * 100
	  )
	$0


# ============================================================
# 7. LIMPIEZA
# ============================================================

snippet quita_na
	${1:datos} |>
	  drop_na(${2:variable})
	$0

snippet cambia_na
	${1:datos} |>
	  mutate(
	    ${2:variable} = replace_na(${2:variable}, ${3:0})
	  )
	$0

snippet recod
	${1:datos} |>
	  mutate(
	    ${2:variable} = recode(
	      ${2:variable},
	      "${3:valor_antiguo}" = "${4:valor_nuevo}"
	    )
	  )
	$0

snippet renombra
	${1:datos} |>
	  rename(
	    ${2:nombre_nuevo} = ${3:nombre_antiguo}
	  )
	$0


# ============================================================
# 8. UNIR BASES
# ============================================================

snippet une
	${1:resultado} <- left_join(
	  ${2:base_1},
	  ${3:base_2},
	  by = c("${4:id}")
	)
	$0

snippet apila
	${1:resultado} <- bind_rows(
	  ${2:base_1},
	  ${3:base_2}
	)
	$0


# ============================================================
# 9. PRUEBAS ESTADÍSTICAS
# ============================================================

snippet tdos
	t.test(
	  ${1:variable_num} ~ ${2:grupo},
	  data = ${3:datos}
	)
	$0

snippet tuno
	t.test(
	  ${1:datos}$${2:variable_num},
	  mu = ${3:valor}
	)
	$0

snippet propor
	x <- sum(${1:datos}$${2:variable_cat} == "${3:valor}", na.rm = TRUE)
	n <- nrow(${1:datos})
	
	prop.test(
	  x = x,
	  n = n,
	  p = ${4:0.80}
	)
	$0

snippet corr
	cor.test(
	  ${1:datos}$${2:variable_x},
	  ${1:datos}$${3:variable_y}
	)
	$0

snippet corrmat
	${1:matriz_cor} <- ${2:datos} |>
	  select(
	    ${3:variable_1},
	    ${4:variable_2},
	    ${5:variable_3},
	    ${6:variable_4}
	  ) |>
	  cor(use = "complete.obs")
	
	corrplot(
	  ${1:matriz_cor},
	  method = "color",
	  addCoef.col = "black",
	  title = "${7:Matriz de correlaciones}"
	)
	$0

snippet regre
	${1:modelo} <- lm(
	  ${2:variable_y} ~ ${3:variable_x},
	  data = ${4:datos}
	)
	
	summary(${1:modelo})
	$0

snippet histo
	ggplot(${1:datos}, aes(x = ${2:variable_num})) +
	  geom_histogram() +
	  labs(
	    title = "${3:Distribución}",
	    x = "${4:Variable}",
	    y = "Frecuencia"
	  ) +
	  theme_minimal() +
	  theme(
	    plot.title = element_text(hjust = 0.5, face = "bold")
	  )
	$0

	    y = "Cantidad"
	  ) +

	    plot.title = element_text(hjust = 0.5, face = "bold")
	  )
	$0

	  group_by(${3:grupo}) |>

	  )
	
	ggplot(${1:resumen}, aes(x = ${3:grupo}, y = promedio)) +

	    vjust = -0.4

snippet circular
	${1:resumen} <- ${2:datos} |>
	  count(${3:variable_cat}) |>
	  mutate(
	    porcentaje = n / sum(n) * 100
	  )
	
	ggplot(${1:resumen}, aes(x = "", y = porcentaje, fill = ${3:variable_cat})) +
	  geom_col(width = 1) +
	  coord_polar(theta = "y") +
	  geom_text(
	    aes(label = paste0(round(porcentaje, 1), "%")),
	    position = position_stack(vjust = 0.5)
	  ) +
	  labs(
	    title = "${4:Distribución porcentual}",
	    fill = "${5:Categoría}"
	  ) +
	  theme_void() +
	  theme(
	    plot.title = element_text(hjust = 0.5, face = "bold"),
	    legend.position = "bottom"
	  )
	$0
